function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["참여일시", "후보ID", "선택후보", "계열사", "성함", "연락처", "비밀번호"]);
  }
  var data = JSON.parse(e.postData.contents);
  var rows = sheet.getDataRange().getValues();
  var cleanInput = String(data.phone).replace(/\D/g, "");

  for (var i = 1; i < rows.length; i++) {
    var rowPhone = String(rows[i][5]).replace(/\D/g, "");
    if (rowPhone === cleanInput) {
      if (data.action === "update" && String(rows[i][6]) === String(data.pin)) {
        sheet.getRange(i + 1, 1, 1, 5).setValues([[data.timestamp, data.candidateId, data.candidateName, data.company, data.name]]);
        return ContentService.createTextOutput(JSON.stringify({ status: "updated" })).setMimeType(ContentService.MimeType.JSON);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "duplicate" })).setMimeType(ContentService.MimeType.JSON);
    }
  }
  sheet.appendRow([data.timestamp, data.candidateId, data.candidateName, data.company, data.name, data.phone, String(data.pin)]);
  return ContentService.createTextOutput(JSON.stringify({ status: "ok" })).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    result.push({
      timestamp: rows[i][0],
      candidateId: Number(rows[i][1]),
      candidateName: rows[i][2],
      company: rows[i][3],
      name: rows[i][4],
      phone: String(rows[i][5]),
      pin: String(rows[i][6] || "")
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", votes: result })).setMimeType(ContentService.MimeType.JSON);
}
